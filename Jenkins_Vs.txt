pipeline {
    agent{
        label "slavenode"
        }
    environment {
        artifactname = 'vibhuapp'
        artifactrepo = 's3/vibhubucket'
    }

    stages {
        stage('Hello') {
            steps {
                echo "Hello World this app name is ${artifactname} and artifact repo is ${artifactrepo}"
                sh "sleep 10"
            }
        }
        stage('Hola') {
            steps {
                echo "Hola Mundo"
                echo "Hola means hello in Spanish"
                sh "pwd"
                sh "ls"
                sh " echo the repo is ${artifactrepo}"
            }
        }
    }
    post {
        success {
            mail to: 'dsv8066@gmail.com',
            subject: 'Build Successful',
            body: 'Pipeline Built Successfully'            }
        failure {
            mail to: 'dsv8066@gmail.com',
            subject: 'Build Failed',
            body: 'Pipeline Failed'
                }
        }
}
